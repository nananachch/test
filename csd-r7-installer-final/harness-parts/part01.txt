def sha(p:Path):return hashlib.sha256(p.read_bytes()).hexdigest()
def tree(root:Path):return {} if not root.exists() else {p.relative_to(root).as_posix():sha(p) for p in root.rglob('*') if p.is_file()}
def require(x,msg):
    if not x: raise AssertionError(msg)

def make_package(case:Path):
    package=case/'package'; (package/'TOOLS').mkdir(parents=True)
    shutil.copy2(INSTALLER,package/'TOOLS/install_or_update_codex_project.py')
    payload=package/'PAYLOAD'/PROJECT/'Latest_Source'

    (payload/'SOURCE/main.cs').write_text('R7 synthetic exact-installer Windows test',encoding='utf-8')
    (payload/'REPORTS').mkdir();(payload/'REPORTS/proof.txt').write_text('proof',encoding='utf-8')
    plug=payload/'_BUNDLE/INSTALL/Expanded_Plugin';plug.mkdir(parents=True)
    independent=package/'PLUGIN_FILES';independent.mkdir()
    contents={
      'CrescentSetupDirector.dll':b'MZ synthetic R7 dll bytes',
      'CrescentSetupDirector.deps.json':b'{"runtimeTarget":{"name":"net10.0-windows"}}',
      'CrescentSetupDirector.json':b'{"AssemblyVersion":"0.2.11.0","DalamudApiLevel":15}',
    }
    for n,data in contents.items():
      (plug/n).write_bytes(data);(independent/n).write_bytes(data)
    (payload/'_BUNDLE/INSTALL/Previous').mkdir(parents=True);(payload/'_BUNDLE/INSTALL/Previous/README.txt').write_text('none')
    man=package/'MANIFEST';man.mkdir()

    plugin_files={p.name:sha(p) for p in independent.iterdir() if p.is_file()}
    (man/'payload-sha256.json').write_text(json.dumps({'files':payload_files},indent=2),encoding='utf-8')
    (man/'plugin-files-sha256.json').write_text(json.dumps({'files':plugin_files},indent=2),encoding='utf-8')
    return package

def run(package:Path, root:Path, **extra):
    env=os.environ.copy();env['CODEX_UPDATE_TEST_ROOT']=str(root)
    for k,v in extra.items():env[k]=str(v)
    return subprocess.run([sys.executable,'-I','-B',str(package/'TOOLS/install_or_update_codex_project.py')],env=env,text=True,capture_output=True)

def seed_old(root:Path, marker='old'):
    t=target(root);(t/'_BUNDLE/INSTALL/Expanded_Plugin').mkdir(parents=True,exist_ok=True);(t/'manual/cache').mkdir(parents=True)
    (t/'old.txt').write_text('old '+marker);(t/'settings.json').write_text('{"unknown":true}');(t/'manual/cache/data.bin').write_bytes(b'user-data')
    for n in ['CrescentSetupDirector.dll','CrescentSetupDirector.deps.json','CrescentSetupDirector.json']:(t/'_BUNDLE/INSTALL/Expanded_Plugin'/n).write_text('old '+n)
    return tree(t)
