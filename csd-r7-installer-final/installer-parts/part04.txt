    write_result("SUCCESS", detail, log_path)
    emit(f"[SUCCESS] {detail}")
    emit(f"最終DLL: {final_plugin / DLL_NAME}")
    emit(f"ログ: {log_path}")
    open_path(final_plugin)
    return 0


if __name__ == "__main__":
    try:
        exit_code = main()
    except Exception as exc:
        details = f"{type(exc).__name__}: {exc}"
        try:
            emit(f"詳細: {details}")
            if LOG is not None:
                LOG.write(traceback.format_exc())
                LOG.flush()
            log_candidate = Path(getattr(LOG, "name", ROOT / "LOGS" if ROOT else "LOGS"))
            write_result("FAILED", details, log_candidate if log_candidate.is_file() else None)
            print(f"[FAILED] {DISPLAY_NAME}の処理に失敗しました。\n原因: {details}\nログ: {log_candidate}", flush=True)
            if log_candidate.is_file():
                open_log(log_candidate)
        finally:
            if LOG is not None:
                LOG.close()
        raise SystemExit(1)
    else:
        if LOG is not None:
            LOG.close()
        raise SystemExit(exit_code)
